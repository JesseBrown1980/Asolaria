const mqtt = require("mqtt");
const os = require("os");

const BROKER = "mqtt://127.0.0.1:18883";
const DEVICE_ID = "jesse_phone";
const VIEWER = "falcon";
const TOPIC_PREFIX = "asolaria/nodes/" + DEVICE_ID;

const client = mqtt.connect(BROKER, {
  clientId: DEVICE_ID,
  clean: true,
  keepalive: 30
});

client.on("connect", () => {
  console.log("[MQTT] Connected to colony broker as " + DEVICE_ID);

  // Subscribe to commands
  client.subscribe("asolaria/nodes/" + DEVICE_ID + "/command/#");
  client.subscribe("asolaria/broadcast/#");

  // Publish presence
  const presence = JSON.stringify({
    nodeId: DEVICE_ID,
    viewer: VIEWER,
    type: "phone",
    model: "SM-S721U1",
    role: "super_master",
    capabilities: ["termux", "node", "python", "adb"],
    at: new Date().toISOString()
  });
  client.publish(TOPIC_PREFIX + "/runtime/presence", presence, { retain: true });

  // Heartbeat every 30s
  setInterval(() => {
    client.publish(TOPIC_PREFIX + "/runtime/health", JSON.stringify({
      ok: true,
      nodeId: DEVICE_ID,
      uptime: process.uptime(),
      mem: process.memoryUsage().rss,
      at: new Date().toISOString()
    }));
  }, 30000);
});

client.on("message", (topic, message) => {
  console.log("[MQTT] " + topic + ": " + message.toString().slice(0, 200));
});

client.on("error", (err) => {
  console.error("[MQTT] Error:", err.message);
});

console.log("[MQTT] Phone agent starting for " + DEVICE_ID + "...");
